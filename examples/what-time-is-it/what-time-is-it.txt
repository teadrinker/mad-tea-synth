rot = (p:[], id1, id2, a) => 
    cosa = cos(a) ; sina = sin(a)
    tmp    = p[id1] * cosa - p[id2] * sina
    p[id2] = p[id2] * cosa + p[id1] * sina
    p[id1] = tmp


do3d = (p:[], r1, r2, r3, w=proj_w, h=proj_h, aspect=proj_aspect, mul=0.01, dist=2.7, xoffset = proj_xoffset) => 
    rot(p, 0, 2, r1)
    rot(p, 1, 2, r2)
    rot(p, 0, 1, r3)
    x = p[0] * mul
    y = p[1] * mul
    z = p[2] * mul + dist
    return [w*((x/z) + 0.5) + xoffset, aspect*h*y/z + h*0.5] 


rsmul = (x, a) => a*x / (2.0 * a*x - a - x + 1.0) 

warp_offset = a => rsmul(a, 0.792) * 0.26795  

rhythmic_time_warp = (x:fx16, amount, beatlength = 0.1, symmetric = 0) => 
    a = beatlength/3.5
    sym = 0.0
    if symmetric > 0
        sym = - amount * a + amount/2 + amount * (smootherstep(-a*2.5, a*2.5, frac(x+0.5) - 0.5) - frac(x+0.5))
        if symmetric == 1.0
            return x + sym
    y = frac(x + warp_offset(a));
    y = ( 1.0 - (1 + y/a) * exp(-y/a) ) * (1 - y);
    return x + amount * y * (1-symmetric) + symmetric * sym;


filtered_superpw = (x, pw, f) => 
    x = frac(x)
    t0 = 0.1875
    t1 = t0 + pw * 1 / 8
    t2 = t0 + pw * 2 / 8
    t3 = t0 + pw * 3 / 8
    t4 = t0 + pw * 4 / 8
    t5 = 1.0 - t0
    o  = linearstep(t0 - f, t0 + f, x) - linearstep(t5 - f, t5 + f, x) 
    o -= linearstep(t1 - f, t1 + f, x) - linearstep(t2 - f, t2 + f, x)
    o -= linearstep(t3 - f, t3 + f, x) - linearstep(t4 - f, t4 + f, x)
    return o - 0.5
